<script>
const SUPABASE_URL = "https://cmzbiyscsalzopsdydtg.supabase.co";
const SUPABASE_ANON_KEY = "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6ImNtemJpeXNjc2Fsem9wc2R5ZHRnIiwicm9sZSI6ImFub24iLCJpYXQiOjE3ODU1ODEwODMsImV4cCI6MjEwMTE1NzA4M30.tuNc5Yftdi-402RCEojcbeiDszaJcfEewqr9UADffHU";

const sb = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

let products = [];
let activeCat = "همه";
let query = "";

async function loadProducts(){
  const grid = document.getElementById('grid');
  grid.innerHTML = `<div class="empty"><div class="big">⏳</div>در حال بارگذاری محصولات...</div>`;

  const { data, error } = await sb.from('products').select('*').order('created_at', {ascending:false});

  if(error){
    grid.innerHTML = `<div class="empty"><div class="big">⚠️</div>خطا در بارگذاری محصولات</div>`;
    return;
  }

  products = (data || []).map(p => ({
    id: p.id, name: p.name, code: p.code, cat: p.category, price: p.price,
    shop: p.shop_name || 'بازارچه', img: p.image_url, emoji: '📦',
    inStock: p.in_stock !== false,
    t1:'#E3E6E1', t2:'#C2C9C2'
  }));

  renderChips();
  renderGrid();
}

const chipRow = document.getElementById('chipRow');
const grid = document.getElementById('grid');
const metaLine = document.getElementById('metaLine');
const searchInput = document.getElementById('searchInput');

function renderChips(){
  const categories = ["همه", ...Array.from(new Set(products.map(p=>p.cat)))];
  chipRow.innerHTML = categories.map(c =>
    `<div class="chip ${c===activeCat?'active':''}" data-cat="${c}">${c}</div>`
  ).join('');
  chipRow.querySelectorAll('.chip').forEach(el=>{
    el.addEventListener('click', ()=>{
      activeCat = el.dataset.cat;
      renderChips();
      renderGrid();
    });
  });
}

function fmtPrice(n){
  return n.toLocaleString('fa-IR');
}

function renderGrid(){
  const filtered = products.filter(p=>{
    const matchCat = activeCat==="همه" || p.cat===activeCat;
    const matchQuery = query.trim()==="" ||
      p.name.includes(query) || p.cat.includes(query) || p.shop.includes(query) || (p.code||'').toLowerCase().includes(query.toLowerCase());
    return matchCat && matchQuery;
  });

  metaLine.textContent = `${filtered.length.toLocaleString('fa-IR')} کالا پیدا شد`;

  if(filtered.length===0){
    grid.innerHTML = `<div class="empty"><div class="big">🔍</div>هیچ کالایی پیدا نشد<br>عبارت دیگری را امتحان کنید</div>`;
    return;
  }

  grid.innerHTML = filtered.map(p => `
    <div class="card">
      <div class="thumb" style="--tint1:${p.t1};--tint2:${p.t2}; ${!p.inStock ? 'filter:grayscale(0.6);opacity:0.7;' : ''}">
        ${p.img ? `<img src="${p.img}" alt="${p.name}" style="width:100%;height:100%;object-fit:cover;" onerror="this.style.display='none'; this.nextElementSibling.style.display='flex';">
        <span style="display:none; width:100%; height:100%; align-items:center; justify-content:center; font-size:38px;">📦</span>` : (p.emoji || '📦')}
        <span class="shopbadge">${p.shop}</span>
        ${!p.inStock ? '<span style="position:absolute;top:8px;left:8px;background:#A32638;color:#fff;font-size:10px;font-weight:700;padding:3px 8px;border-radius:999px;">ناموجود</span>' : ''}
      </div>
      <div class="info">
        <div class="pname">${p.name}${p.code ? ` <span style="color:var(--pomegranate);font-weight:800;">[${p.code}]</span>` : ''}</div>
        <div class="pcat">${p.cat}</div>
        <div class="prow">
          <div class="price">${fmtPrice(p.price)} <small>تومان</small></div>
          ${p.inStock
            ? `<button class="buybtn" onclick="orderProduct('${p.name.replace(/'/g,"\\'")}', '${(p.code||'').replace(/'/g,"\\'")}')">سفارش</button>`
            : `<button class="buybtn" disabled style="opacity:0.5;">ناموجود</button>`}
        </div>
      </div>
    </div>
  `).join('');
}

searchInput.addEventListener('input', (e)=>{
  query = e.target.value;
  renderGrid();
});

function orderProduct(name, code){
  const codeText = code ? ` (کد: ${code})` : '';
  const msg = encodeURIComponent(`سلام، می‌خوام این محصول رو سفارش بدم: ${name}${codeText}`);
  window.open(`https://t.me/Sharifi190?text=${msg}`, '_blank');
}

function toggleHadis(){
  document.getElementById('hadisWindow').classList.toggle('open');
}

async function sendHadis(){
  const input = document.getElementById('hadisInput');
  const text = input.value.trim();
  if(!text) return;

  const msgs = document.getElementById('hadisMsgs');
  msgs.innerHTML += `<div class="hmsg user">${text}</div>`;
  input.value = '';
  msgs.scrollTop = msgs.scrollHeight;

  const loadingId = 'loading-' + Date.now();
  msgs.innerHTML += `<div class="hmsg loading" id="${loadingId}">در حال تایپ...</div>`;
  msgs.scrollTop = msgs.scrollHeight;

  try{
    const res = await fetch(`${SUPABASE_URL}/functions/v1/hadis-chat`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${SUPABASE_ANON_KEY}`
      },
      body: JSON.stringify({ message: text })
    });
    const data = await res.json();
    document.getElementById(loadingId).remove();
    msgs.innerHTML += `<div class="hmsg bot">${data.reply || 'ببخشید، مشکلی پیش اومد.'}</div>`;
  } catch(err){
    document.getElementById(loadingId).remove();
    msgs.innerHTML += `<div class="hmsg bot">ارتباط برقرار نشد، دوباره امتحان کن.</div>`;
  }
  msgs.scrollTop = msgs.scrollHeight;
}

loadProducts();
