function getRatingHtml(id){
  const r = ratingsMap[id];
  if(!r || r.count===0) return `<span class="stars">☆☆☆☆☆ <span class="cnt">بدون امتیاز</span></span>`;
  const avg = Math.round(r.sum / r.count);
  const stars = '★'.repeat(avg) + '☆'.repeat(5-avg);
  return `<span class="stars">${stars} <span class="cnt">(${r.count.toLocaleString('fa-IR')})</span></span>`;
}

function sortProducts(list){
  const mode = document.getElementById('sortSelect') ? document.getElementById('sortSelect').value : 'new';
  const copy = [...list];
  if(mode==='bestseller') copy.sort((a,b)=> b.orderCount - a.orderCount);
  else if(mode==='priceAsc') copy.sort((a,b)=> a.price - b.price);
  else if(mode==='priceDesc') copy.sort((a,b)=> b.price - a.price);
  return copy;
}

async function orderProduct(name, code){
  const codeText = code ? ` (کد: ${code})` : '';
  const msg = encodeURIComponent(`سلام، می‌خوام این محصول رو سفارش بدم: ${name}${codeText}`);
  window.open(`https://t.me/Sharifi190?text=${msg}`, '_blank');

}

/* ---------- Cart ---------- */
function saveCart(){ localStorage.setItem('bazarcheh_cart', JSON.stringify(cart)); }

function addToCart(id){
  const existing = cart.find(c => c.id===id);
  if(existing) existing.qty += 1;
  else cart.push({ id, qty: 1 });
  saveCart();
  updateBadges();
  if(document.getElementById('cartDrawer').classList.contains('open')) renderCartDrawer();
}

function changeQty(id, delta){
  const item = cart.find(c => c.id===id);
  if(!item) return;
  item.qty += delta;
  if(item.qty <= 0) cart = cart.filter(c => c.id!==id);
  saveCart();
  updateBadges();
  renderCartDrawer();
}

function removeFromCart(id){
  cart = cart.filter(c => c.id!==id);
  saveCart();
  updateBadges();
  renderCartDrawer();
}

function updateBadges(){
  const cartCount = cart.reduce((s,c)=>s+c.qty, 0);
  const cartBadge = document.getElementById('cartBadge');
  cartBadge.textContent = cartCount;
  cartBadge.style.display = cartCount>0 ? 'flex' : 'none';

  const wishBadge = document.getElementById('wishBadge');
  wishBadge.textContent = wishlist.length;
  wishBadge.style.display = wishlist.length>0 ? 'flex' : 'none';
}

function openCart(){
  document.getElementById('cartDrawer').classList.add('open');
  document.getElementById('cartOverlay').classList.add('open');
  renderCartDrawer();
}
function closeCart(){
  document.getElementById('cartDrawer').classList.remove('open');
  document.getElementById('cartOverlay').classList.remove('open');
}

function cartTotalRaw(){
  return cart.reduce((sum, c) => {
    const p = products.find(x => x.id===c.id);
    return p ? sum + p.price * c.qty : sum;
  }, 0);
}

function renderCartDrawer(){
  const body = document.getElementById('cartBody');
  const foot = document.getElementById('cartFoot');
  if(cart.length===0){
    body.innerHTML = `<div class="empty-drawer">سبد خریدت خالیه 🛍</div>`;
    foot.style.display = 'none';
    return;
  }
  foot.style.display = 'block';
  body.innerHTML = cart.map(c => {
    const p = products.find(x => x.id===c.id);
    if(!p) return '';
    return `
      <div class="cart-item">
        <img src="${p.img||''}" onerror="this.style.background='#eee'; this.src='';">
        <div class="ci-info">
          <div class="ci-name">${p.name}</div>
          <div class="ci-price">${fmtPrice(p.price * c.qty)} تومان</div>
          <div class="qty-row">
            <button onclick="changeQty('${p.id}', -1)">−</button>
            <span>${c.qty.toLocaleString('fa-IR')}</span>
            <button onclick="changeQty('${p.id}', 1)">+</button>
          </div>
        </div>
        <button class="cart-remove" onclick="removeFromCart('${p.id}')">✕</button>
      </div>
    `;
  }).join('');

  let total = cartTotalRaw();
  if(appliedCoupon) total = Math.round(total * (1 - appliedCoupon.percent/100));
  document.getElementById('cartTotal').textContent = fmtPrice(total) + ' تومان';
}

async function applyCoupon(){
  const code = document.getElementById('couponInput').value.trim().toUpperCase();
  const msgEl = document.getElementById('couponMsgShop');
  if(!code){ msgEl.textContent=''; appliedCoupon=null; renderCartDrawer(); return; }

  const { data, error } = await sb.from('coupons').select('*').eq('code', code).eq('active', true).maybeSingle();
  if(error || !data){
    appliedCoupon = null;
    msgEl.style.color = 'var(--pomegranate)';
    msgEl.textContent = 'کد تخفیف معتبر نیست';
  } else {
    appliedCoupon = data;
    msgEl.style.color = '#2A6B3B';
    msgEl.textContent = `${data.percent}٪ تخفیف اعمال شد ✅`;
  }
  renderCartDrawer();
}

function checkoutCart(){
  if(cart.length===0) return;
  let lines = cart.map(c => {
    const p = products.find(x => x.id===c.id);
    if(!p) return '';
    const codeText = p.code ? ` (کد: ${p.code})` : '';
    return `- ${p.name}${codeText} × ${c.qty}`;
  }).filter(Boolean).join('\n');

  let total = cartTotalRaw();
  let discountLine = '';
  if(appliedCoupon){
    total = Math.round(total * (1 - appliedCoupon.percent/100));
    discountLine = `\nکد تخفیف: ${appliedCoupon.code} (${appliedCoupon.percent}٪)`;
  }

  const msg = encodeURIComponent(`سلام، می‌خوام این سفارش رو ثبت کنم:\n${lines}${discountLine}\n\nمجموع: ${fmtPrice(total)} تومان`);
  window.open(`https://t.me/Sharifi190?text=${msg}`, '_blank');

  cart.forEach(c => sb.rpc('increment_order_count', { pid: c.id }).then(()=>{}).catch(()=>{}));
}

async function rateProduct(id, rating){
  await sb.from('product_ratings').insert({ product_id: id, rating });
  const { data: ratingsData } = await sb.from('product_ratings').select('product_id, rating');
  ratingsMap = {};
  (ratingsData || []).forEach(r => {
    if(!ratingsMap[r.product_id]) ratingsMap[r.product_id] = { sum:0, count:0 };
    ratingsMap[r.product_id].sum += r.rating;
    ratingsMap[r.product_id].count += 1;
  });
  openProductDetail(id);
  renderGrid();
}

async function sendHadis(){
  const input = document.getElementById('hadisInput');
  const text = input.value.trim();
  if(!text) return;

  const msgs = document.getElementById('hadisMsgs');
  msgs.innerHTML += `<div class="hmsg user">${text}</div>`;
  input.value = '';
  msgs.scrollTop = msgs.scrollHeight;

  const loadingId = 'loading-' + Date.now();
  msgs.innerHTML += `<div class="hmsg loading" id="${loadingId}">در حال تایپ...</div>`;
  msgs.scrollTop = msgs.scrollHeight;

  try{
    const res = await fetch(`${SUPABASE_URL}/functions/v1/hadis-chat`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${SUPABASE_ANON_KEY}`
      },
      body: JSON.stringify({ message: text })
    });
    const data = await res.json();
    document.getElementById(loadingId).remove();
    msgs.innerHTML += `<div class="hmsg bot">${data.reply || 'ببخشید، مشکلی پیش اومد.'}</div>`;
  } catch(err){
    document.getElementById(loadingId).remove();
    msgs.innerHTML += `<div class="hmsg bot">ارتباط برقرار نشد، دوباره امتحان کن.</div>`;
  }
  msgs.scrollTop = msgs.scrollHeight;
}
